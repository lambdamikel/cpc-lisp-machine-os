(DEFUN FIB (N) (COND ((LESSP N 2) N)
(T (PLUS (FIB (DIFFERENCE N 1)) (FIB (DIFFERENCE N 2))))))
(DEFUN TAK (X Y Z) (COND ((NOT (LESSP Y X)) Z)
(T (TAK (TAK (DIFFERENCE X 1) Y Z) (TAK (DIFFERENCE Y 1) Z X)
(TAK (DIFFERENCE Z 1) X Y)))))
(DEFUN DVAUX (A) (LIST '/ (DV A) A))
(DEFUN DV (A) (COND ((ATOM A) (COND ((EQ A 'X) 1) (T 0)))
((EQ (CAR A) '+) (CONS '+ (MAPCAR (CDR A) 'DV)))
((EQ (CAR A) '- ) (CONS '- (MAPCAR (CDR A) 'DV)))
((EQ (CAR A) '*) (LIST '* A (CONS '+ (MAPCAR (CDR A) 'DVAUX))))
((EQ (CAR A) '/) (LIST '- (LIST '/ (DV (CADR A)) (CADDR A))
(LIST '/ (CADR A) (LIST '* (CADDR A) (CADDR A) (DV (CADDR A))))))
(T 'ERROR)))
